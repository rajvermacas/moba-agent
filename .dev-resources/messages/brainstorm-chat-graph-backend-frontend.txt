          // Transform data based on decision

          RETURN state

  Simple Data Transformer:

  FUNCTION transform_data(results, chart_type, x_field, y_fields):
      """
      Single transformer function - no classes needed
      """

      TRY:
          IF chart_type == "pie":
              # Simple aggregation for pie
              IF len(results) > 10:
                  # Take top 10 + "Others"
                  top_10 = results[:10]
                  others_value = sum(r[y_fields[0]] for r in results[10:])
                  RETURN top_10 + [{"name": "Others", "value":
  others_value}]
              RETURN [{"name": r[x_field], "value": r[y_fields[0]]} for r
  in results]

          ELIF chart_type == "scatter":
              # Direct mapping for scatter
              RETURN [{"x": r[x_field], "y": r[y_fields[0]]} for r in
  results]

          ELIF chart_type in ["bar", "line", "area"]:
              # Standard format for bar/line/area
              RETURN results  # Already in correct format

          ELIF chart_type == "heatmap":
              # Flatten for heatmap
              RETURN [
                  {"x": r[x_field], "y": r[y_fields[0]], "value":
  r[y_fields[1]]}
                  for r in results
              ]

      EXCEPT Exception as e:
          LOG.error(f"Transform failed: {e}")
          RETURN None

  Graph Generation Function with Heavy Logging:

  FUNCTION generate_graph_from_results(sql_query, query_results):
      LOG.info("="*50)
      LOG.info("Starting graph generation")
      LOG.info(f"SQL Query: {sql_query}")
      LOG.info(f"Result rows: {len(query_results)}")

      # Validation checks
      IF not query_results:
          LOG.warning("No query results - returning null graph")
          RETURN None

      IF len(query_results) < 2:
          LOG.warning(f"Only {len(query_results)} rows - too few for 
  visualization")
          RETURN None

      # Analyze data structure
      columns = list(query_results[0].keys())
      LOG.info(f"Columns found: {columns}")

      column_types = {}
      for col in columns:
          sample_value = query_results[0][col]
          column_types[col] = type(sample_value).__name__
          LOG.debug(f"Column '{col}': type={column_types[col]}, 
  sample={sample_value}")

      # Prepare LLM prompt
      prompt = build_visualization_prompt(sql_query, columns,
  column_types, query_results[:5])
      LOG.debug(f"LLM Prompt: {prompt[:500]}...")  # Log first 500 chars

      # Call LLM
      TRY:
          LOG.info("Calling LLM for graph recommendation")
          llm_response = agent.llm.invoke(prompt, response_format={"type":
   "json"})
          LOG.info(f"LLM Response: {llm_response}")

          recommendation = json.loads(llm_response)
          LOG.info(f"LLM recommends: {recommendation['type']} chart")
          LOG.debug(f"Field mappings: x={recommendation.get('x_field')}, 
  y={recommendation.get('y_fields')}")

      EXCEPT Exception as e:
          LOG.error(f"LLM call failed: {str(e)}")
          LOG.error(f"Stack trace: {traceback.format_exc()}")
          RETURN None

      # Check if LLM thinks it's visualizable
      IF not recommendation.get('can_visualize', False):
          LOG.warning(f"LLM determined data not suitable for 
  visualization. Reason: {recommendation.get('reason')}")
          RETURN None

      # Transform data
      LOG.info(f"Transforming data for {recommendation['type']} chart")
      TRY:
          transformed_data = transform_for_recharts(
              query_results,
              recommendation['type'],
              recommendation.get('x_field'),
              recommendation.get('y_fields', [])
          )
          LOG.info(f"Transformation successful - {len(transformed_data)} 
  data points")
          LOG.debug(f"Sample transformed data: {transformed_data[:2]}")

      EXCEPT Exception as e:
          LOG.error(f"Data transformation failed: {str(e)}")
          LOG.error(f"Stack trace: {traceback.format_exc()}")
          RETURN None

      # Build graph object
      graph = {
          "type": recommendation['type'],
          "data": transformed_data,
          "config": {
              "xKey": recommendation.get('x_field'),
              "yKeys": recommendation.get('y_fields', []),
              "title": recommendation.get('title', ''),
              "xLabel": recommendation.get('x_label', ''),
              "yLabel": recommendation.get('y_label', '')
          }
      }

      LOG.debug(f"Graph object created: {json.dumps(graph, 
  indent=2)[:500]}...")

      # Validate
      LOG.info("Validating graph object")
      validation_result = validate_graph_object(graph)

      IF not validation_result['valid']:
          LOG.error(f"Graph validation failed: 
  {validation_result['errors']}")
          LOG.error(f"Invalid graph object: {json.dumps(graph)}")
          RETURN None

      LOG.info("Graph validation successful")
      LOG.info("="*50)
      RETURN graph

  Data Transformation with Logging:

  FUNCTION transform_for_recharts(results, chart_type, x_field, y_fields):
      LOG.info(f"Starting transformation for {chart_type}")
      LOG.debug(f"X field: {x_field}, Y fields: {y_fields}")
      LOG.debug(f"Input data count: {len(results)}")

      TRY:
          IF chart_type == "pie":
              LOG.info("Transforming for pie chart")

              # Aggregate if more than 10 categories
              IF len(results) > 10:
                  LOG.warning(f"Found {len(results)} categories - limiting
   to top 10 + Others")
                  top_10 = results[:10]
                  others_sum = sum(float(r[y_fields[0]]) for r in
  results[10:])
                  LOG.debug(f"Others category sum: {others_sum}")

                  transformed = [
                      {"name": str(r[x_field]), "value":
  float(r[y_fields[0]])}
                      for r in top_10
                  ]
                  transformed.append({"name": "Others", "value":
  others_sum})
              ELSE:
                  transformed = [
                      {"name": str(r[x_field]), "value":
  float(r[y_fields[0]])}
                      for r in results
                  ]

              LOG.info(f"Pie chart data ready: {len(transformed)} slices")
              RETURN transformed

          ELIF chart_type == "scatter":
              LOG.info("Transforming for scatter plot")
              transformed = []

              FOR i, row in enumerate(results):
                  TRY:
                      point = {
                          "x": float(row[x_field]),
                          "y": float(row[y_fields[0]])
                      }
                      transformed.append(point)
                  EXCEPT (ValueError, TypeError) as e:
                      LOG.warning(f"Row {i} has non-numeric data: {e}")

              LOG.info(f"Scatter plot data ready: {len(transformed)} 
  points")
              RETURN transformed

          ELIF chart_type in ["bar", "line", "area"]:
              LOG.info(f"Transforming for {chart_type} chart")

              # Keep original structure for Recharts
              # Just ensure y_fields values are numeric
              transformed = []
              FOR i, row in enumerate(results):
                  clean_row = {x_field: row[x_field]}

                  FOR y_field in y_fields:
                      TRY:
                          clean_row[y_field] = float(row[y_field])
                      EXCEPT (ValueError, TypeError):
                          LOG.warning(f"Row {i}, field {y_field} is 
  non-numeric: {row[y_field]}")
                          clean_row[y_field] = 0

                  transformed.append(clean_row)

              LOG.info(f"{chart_type} chart data ready: {len(transformed)}
   points")
              RETURN transformed

          ELIF chart_type == "heatmap":
              LOG.info("Transforming for heatmap")
              transformed = []

              FOR row in results:
                  transformed.append({
                      "x": str(row[x_field]),
                      "y": str(row[y_fields[0]]),
                      "value": float(row[y_fields[1]] if len(y_fields) > 1
   else row[y_fields[0]])
                  })

              LOG.info(f"Heatmap data ready: {len(transformed)} cells")
              RETURN transformed

          ELSE:
              LOG.error(f"Unknown chart type: {chart_type}")
              RETURN None

      EXCEPT Exception as e:
          LOG.error(f"Transformation failed: {str(e)}")
          LOG.error(f"Stack trace: {traceback.format_exc()}")
          RETURN None

  Simple Validation with Logging:

  FUNCTION validate_graph_object(graph):
      LOG.info("Starting graph validation")
      errors = []

      # Check structure
      IF not graph:
          errors.append("Graph object is None")
          LOG.error("Graph object is None")

      IF not graph.get("type"):
          errors.append("Missing graph type")
          LOG.error("Missing graph type")

      IF not graph.get("data"):
          errors.append("Missing data array")
          LOG.error("Missing data array")

      IF graph.get("data") and len(graph["data"]) == 0:
          errors.append("Data array is empty")
          LOG.error("Data array is empty")

      # Type-specific validation
      IF graph.get("type") and graph.get("data") and len(graph["data"]) >
  0:
          first_row = graph["data"][0]
          LOG.debug(f"Validating {graph['type']} with first row: 
  {first_row}")

          IF graph["type"] == "pie":
              IF "name" not in first_row or "value" not in first_row:
                  errors.append("Pie chart missing 'name' or 'value' 
  fields")
                  LOG.error(f"Pie chart validation failed. First row: 
  {first_row}")

          ELIF graph["type"] == "scatter":
              IF "x" not in first_row or "y" not in first_row:
                  errors.append("Scatter plot missing 'x' or 'y' fields")
                  LOG.error(f"Scatter plot validation failed. First row: 
  {first_row}")

          ELIF graph["type"] in ["bar", "line", "area"]:
              config = graph.get("config", {})
              x_key = config.get("xKey")
              y_keys = config.get("yKeys", [])

              IF not x_key or x_key not in first_row:
                  errors.append(f"Missing or invalid xKey: {x_key}")
                  LOG.error(f"xKey '{x_key}' not found in data")

              FOR y_key in y_keys:
                  IF y_key not in first_row:
                      errors.append(f"Missing yKey: {y_key}")
                      LOG.error(f"yKey '{y_key}' not found in data")

      IF errors:
          LOG.error(f"Validation failed with {len(errors)} errors")
          RETURN {"valid": False, "errors": errors}

      LOG.info("Validation successful")
      RETURN {"valid": True, "errors": []}
